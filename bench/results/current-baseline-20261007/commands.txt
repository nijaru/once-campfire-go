# Common Docker invocation used for each Python command below:

  -v "$PWD:/src:ro" -v "$PWD/.cache:/cache" \
  -v "$PWD/.cache/performance-current-20261007/work:/src/bench/.work" \
  -w /src -e CGO_ENABLED=1 -e GOMAXPROCS=4 \
  campfire-compare-toolchain python3 bench/application \
  --rust-root /src/reference --seed /cache/parity-seeds/default \
  --loadgen /cache/linux/loadgen \

  --out /cache/performance-current-20261007/capacity \
  --apps rust go --reps 3 --seconds 10 --concurrency 16 --gzip 1 \
