package front
import("sort";"strconv";"strings";"net/http";"net/http/httptest";"math/rand";"testing")
func probeOriginalEncoding(header string) string {
	type item struct {
		name       string
		q          float64
		preference int
	}
	var accepts []item
	for _, part := range strings.Split(header, ",") {
		part = strings.TrimSpace(part)
		if part == "" {
			continue
		}
		name, param, _ := strings.Cut(part, ";")
		name = strings.TrimSpace(name)
		q := 1.0
		param = strings.TrimSpace(param)
		if strings.HasPrefix(param, "q=") {
			value := strings.TrimPrefix(param, "q=")
			end := 0
			for end < len(value) && (value[end] >= '0' && value[end] <= '9' || value[end] == '.') {
				end++
			}
			if end > 0 {
				q, _ = strconv.ParseFloat(value[:end], 64)
			}
		}
		p := 2
		if name == "gzip" {
			p = 0
		} else if name == "identity" {
			p = 1
		}
		accepts = append(accepts, item{name, q, p})
		if len(accepts) == 16 {
			break
		}
	}
	var expanded []item
	wildcard := false
	for _, item := range accepts {
		if item.name != "*" {
			expanded = append(expanded, item)
			continue
		}
		if wildcard {
			continue
		}
		wildcard = true
		for _, name := range []string{"gzip", "identity"} {
			found := false
			for _, v := range accepts {
				found = found || v.name == name
			}
			if !found {
				copy := item
				copy.name = name
				expanded = append(expanded, copy)
			}
		}
	}
	rejected := map[string]bool{}
	hasIdentity := false
	for _, item := range expanded {
		if item.q == 0 {
			rejected[item.name] = true
		}
		hasIdentity = hasIdentity || item.name == "identity"
	}
	sort.SliceStable(expanded, func(i, j int) bool {
		if expanded[i].q == expanded[j].q {
			return expanded[i].preference < expanded[j].preference
		}
		return expanded[i].q > expanded[j].q
	})
	if !hasIdentity {
		expanded = append(expanded, item{name: "identity"})
	}
	for _, item := range expanded {
		if !rejected[item.name] && (item.name == "gzip" || item.name == "identity") {
			return item.name
		}
	}
	return ""
}
func probeOriginalPublicEncoding(r *http.Request) string {
	if r.Method == "HEAD" {
		return ""
	}
	quality := func(name string) float64 {
		for _, part := range strings.Split(r.Header.Get("Accept-Encoding"), ",") {
			pieces := strings.Split(part, ";")
			if !strings.EqualFold(strings.TrimSpace(pieces[0]), name) {
				continue
			}
			q := 1.0
			for _, p := range pieces[1:] {
				if value, ok := strings.CutPrefix(strings.TrimSpace(p), "q="); ok {
					q, _ = strconv.ParseFloat(value, 64)
					q = min(1, max(0, q))
				}
			}
			return q
		}
		return 0
	}
	gz, zs := quality("gzip"), quality("zstd")
	if zs > 0 && zs >= gz {
		return "zstd"
	}
	if gz > 0 {
		return "gzip"
	}
	return ""
}

func TestNegotiationProbe(t *testing.T) {
 r:=rand.New(rand.NewSource(9))
 names:=[]string{"gzip","identity","*","unknown","zstd","GZIP","","br"}
 params:=[]string{"",";q=0",";q=.5",";q=1",";q=1.5",";q=abc",";q=.",";q=0.5.1",";Q=0",";q=NaN",";x=1;q=0",";q=.5;q=0"}
 for n:=0;n<20000;n++ {
  var fields []string
  for j:=0;j<r.Intn(17);j++ {fields=append(fields,names[r.Intn(len(names))]+params[r.Intn(len(params))])}
  header:=strings.Join(fields, ", ")
  if a,b:=encoding(header),probeOriginalEncoding(header);a!=b {t.Fatalf("private %q => %q instead of %q",header,a,b)}
  request:=httptest.NewRequest("GET","/",nil);request.Header.Set("Accept-Encoding",header)
  if a,b:=publicEncoding(request),probeOriginalPublicEncoding(request);a!=b {t.Fatalf("public %q => %q instead of %q",header,a,b)}
 }
}
var probeSelected string
func BenchmarkNegotiationProbe(b *testing.B) {
 const header="gzip, deflate, br"
 b.Run("original",func(b *testing.B){for b.Loop(){probeSelected=probeOriginalEncoding(header)}})
 b.Run("bounded",func(b *testing.B){for b.Loop(){probeSelected=encoding(header)}})
}
