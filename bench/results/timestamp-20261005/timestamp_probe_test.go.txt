package database

import (
	"testing"
	"time"
)

func BenchmarkTimestampProbe(b *testing.B) {
	raw := "2026-10-05 12:34:56.123456"
	b.Run("canonical-scan", func(b *testing.B) {
		b.ReportAllocs()
		var stamp time.Time
		for b.Loop() {
			if err := (timestamp{&stamp}).Scan(raw); err != nil { b.Fatal(err) }
		}
	})
	b.Run("stdlib-parse", func(b *testing.B) {
		b.ReportAllocs()
		for b.Loop() {
			if _, err := time.Parse("2006-01-02 15:04:05.999999999", raw); err != nil { b.Fatal(err) }
		}
	})
}
